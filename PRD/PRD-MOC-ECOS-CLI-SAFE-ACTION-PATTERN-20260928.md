## 5. Architecture d'intégration

```python
# ecos_cli/safe_action_pattern_integration.py
from unified_design.integration_framework import DesignGate

class EcosSafeActionPattern:
    """Pattern PATRON-0 pour ECOS-CLI."""
    
    def __init__(self):
        self.gate = DesignGate("safe-action-pattern", {
            "strict_mode": True,
            "log_violations": True
        })
    
    def verify_before_action(self, action: str, context: dict) -> bool:
        """Vérifie les 7 fonctions + 3 états + invariant."""
        return self.gate.verify({
            "action": action,
            "context": context
        })
```
