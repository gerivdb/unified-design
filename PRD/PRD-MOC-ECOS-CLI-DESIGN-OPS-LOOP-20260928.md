## 5. Architecture d'intégration

```python
# ecos_cli/design_ops_loop_integration.py
from unified_design.integration_framework import DesignOpsLoop

class EcosDesignOpsLoop:
    """Boucle THINK/DO/CHECK pour ECOS-CLI."""
    
    def think(self, command: str, args: dict) -> Thought:
        """Phase THINK : percevoir le contexte, évaluer les risques."""
        return self.loop.think({
            "command": command,
            "args": args
        })
    
    def do(self, thought: Thought) -> ActionResult:
        """Phase DO : exécuter l'action."""
        return self.loop.do(thought)
    
    def check(self, result: ActionResult) -> CheckResult:
        """Phase CHECK : valider le résultat."""
        return self.loop.check(result)
```
