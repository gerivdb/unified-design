## 5. Architecture d'intégration

```python
# kiva_cli/safe_action_gate_integration.py
from unified_design.integration_framework import DesignGate

class KivaSafeActionGate:
    """Gate pré-action pour KIVA-CLI."""
    
    def __init__(self):
        self.gate = DesignGate("safe-action-gate", {
            "strict_mode": True,
            "log_violations": True
        })
    
    def verify_before_command(self, command: str, context: dict) -> bool:
        """Vérifie les préconditions avant exécution d'une commande."""
        return self.gate.verify({
            "command": command,
            "context": context
        })
```
