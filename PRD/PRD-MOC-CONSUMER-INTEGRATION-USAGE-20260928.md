## 4. Script de vérification

```python
# scripts/verify_integration_usage.py
# Vérifie que les modules d'intégration sont importés dans le code métier

import subprocess
import sys
from pathlib import Path

def verify_consumer_usage(consumer: str, design: str) -> dict:
    """Verify that integration module is used in consumer business code."""
    consumer_path = CONSUMER_MAP[consumer]
    slug = design.replace("-", "_")
    
    # Find integration module
    integration_files = list(consumer_path.rglob(f"{slug}_integration.py"))
    if not integration_files:
        return {"status": "MISSING_MODULE", "imports": []}
    
    # Find imports in business code
    result = subprocess.run(
        ["grep", "-r", f"{slug}_integration", str(consumer_path), "--include=*.py"],
        capture_output=True, text=True
    )
    imports = [line for line in result.stdout.split("\n") if line and "test" not in line.lower()]
    
    return {
        "status": "OK" if imports else "NOT_USED",
        "imports": imports,
    }
```
